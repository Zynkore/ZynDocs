## Creating window
```luau
local Window = Library:MakeWindow({
  Title = "Cool title",
  UICorner = false,
  CornerRadius = UDim.new(0, 6)
})
```

<Accordions type="single">
  <Accordion title="Title">
    <p>Window title</p>
    **Default:** `"Missing title"`
  </Accordion>

  <Accordion title="UICorner">
    <p>Window corner</p>
    **Default:** `false`
  </Accordion>

  <Accordion title="CornerRadius">
    <p>Window corner radius</p>
    **Default:** `UDim.new(0, 6)`
  </Accordion>
</Accordions>
